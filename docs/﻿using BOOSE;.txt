using BOOSE;
using BOOSECoreApp.Core.Adapters;
using BOOSECoreApp.Core.Mocks;
using BOOSECoreApp.Core.Variables;
using Boolean = BOOSE.Boolean;

namespace BOOSETests.Variables
{
    /// <summary>
    /// Unit tests for the BOOSE Boolean variable functionality.
    /// </summary>
    [TestClass]
    public class BOOSEBooleanTests
    {
        private StoredProgram? _program;
        private MockGraphicAdapter? _adapter;
        private Boolean? _boolean;

        /// <summary>
        /// Test initialization to set up the StoredProgram and Boolean variable before each test.
        /// </summary>
        [TestInitialize]
        public void Setup()
        {
            _program = new StoredProgram(new MockCanvas());
            _adapter = new MockGraphicAdapter();
            _boolean = new Boolean();
        }

        /// <summary>
        /// Test to verify that a boolean variable can be assigned a value correctly in BOOSE.
        /// </summary>
        [TestMethod]
        public void BoolBOOSE_AssignsCorrectValues()
        {
            _boolean!.Set(_program, "flag1 = true");
            _boolean!.Compile();
            _boolean!.Execute();

            var result = _program!.GetVariable("flag1") as Boolean;
            Assert.AreEqual(1, result!.Value, "Variable value should match the assigned value.");
        }

        /// <summary>
        /// Test to verify that assigning an invalid expression to a boolean variable throws an exception.
        /// </summary>
        [TestMethod]
        public void BoolBOOSE_ThrowsInvalidExpression()
        {
            _boolean!.Set(_program, "flag1 = false");
            _boolean!.Compile();
        }
    }
}
