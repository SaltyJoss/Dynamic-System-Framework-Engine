using BOOSE;
using BOOSECoreApp.Core.Adapters;
using BOOSECoreApp.Core.Base;
using BOOSECoreApp.Core.Interfaces;
using BOOSECoreApp.Core.Mocks;
using System.Drawing;

namespace BOOSETests.Variables
{
    /// <summary>
    /// Unit tests for the BOOSE Array variable functionality.
    /// </summary>
    [TestClass]
    public class BOOSEArrayTests
    {
        /// <summary>
        /// Test to verify that an integer array can be created, compiled, and executed correctly in BOOSE.
        /// </summary>
        [TestMethod]
        public void BOOSEIntArray_CompileAndExecuteTest()
        {
            var canvas = new MockCanvas();
            var program = new StoredProgram(canvas);
            var factory = new CommandFactory();
            var parser = new BOOSE.Parser(factory, program);

            string booseCode = @"
                array int nums 2
                peek nums 2 = 27
            ";

            foreach (var line in booseCode.Split('\n'))
            {
                var trimmed = line.Trim();
                if (string.IsNullOrEmpty(trimmed)) continue;

                ICommand cmd = parser.ParseCommand(trimmed);
                program.Add(cmd);
                cmd.Compile();
                cmd.Execute();
            }

            var nums = (BOOSE.Array)program.GetVariable("nums");

            Assert.AreEqual(27, nums.GetIntArray(2, 0));
        }

        /// <summary>
        /// Test to verify that creating more integer arrays than allowed throws a RestrictionException in BOOSE.
        /// </summary>
        [TestMethod]
        [ExpectedException(typeof(BOOSE.RestrictionException))]
        public void BOOSEIntArray_Restriction_ArrayLimitTest()
        {
            var canvas = new MockCanvas();
            var program = new StoredProgram(canvas);
            var factory = new CommandFactory();
            var parser = new BOOSE.Parser(factory, program);

            string booseCode = @"
                array int arr1 5
                poke arr1 4 = 10
                peek arr2 3 = 20
            ";

            foreach (var line in booseCode.Split('\n').Select(l => l.Trim()).Where(l => !string.IsNullOrEmpty(l)))
            {
                ICommand cmd = parser.ParseCommand(line);
                program.Add(cmd);
                cmd.Compile();
                cmd.Execute(); 
            }
        }

        /// <summary>
        /// Test to verify that an integer array can be created, compiled, and executed correctly in BOOSE.
        /// </summary>
        [TestMethod]
        public void BOOSERealArray_CompileAndExecuteTest()
        {
            var canvas = new MockCanvas();
            var program = new StoredProgram(canvas);
            var factory = new CommandFactory();
            var parser = new BOOSE.Parser(factory, program);

            string booseCode = @"
                array real nums 3
                poke nums 2 = 3.141
                poke nums 1 = 1.618
                poke nums 0 = 2.718 
            ";

            foreach (var line in booseCode.Split('\n'))
            {
                var trimmed = line.Trim();
                if (string.IsNullOrEmpty(trimmed)) continue;

                ICommand cmd = parser.ParseCommand(trimmed);
                program.Add(cmd);
                cmd.Compile();
                cmd.Execute();
            }

            var nums = (BOOSE.Array)program.GetVariable("nums");

            Assert.AreEqual(3.141, nums.GetRealArray(2, 0), "Index 2,0 should match the first 3 digits of pi");
            Assert.AreEqual(1.618, nums.GetRealArray(1, 0), "Index 2,0 should match the first 3 digits of phi");
            Assert.AreEqual(2.718, nums.GetRealArray(0, 0), "Index 2,0 should match the first 3 digits of e");
        }

        /// <summary>
        /// Test to verify that creating more integer arrays than allowed throws a RestrictionException in BOOSE.
        /// </summary>
        [TestMethod]
        [ExpectedException(typeof(BOOSE.RestrictionException))]
        public void BOOSERealArray_Restriction_ArrayLimitTest()
        {
            var canvas = new MockCanvas();
            var program = new StoredProgram(canvas);
            var factory = new CommandFactory();
            var parser = new BOOSE.Parser(factory, program);

            string booseCode = @"
                array real arr3 1
                array real arr2 2
                array real arr1 3 
            ";

            foreach (var line in booseCode.Split('\n').Select(l => l.Trim()).Where(l => !string.IsNullOrEmpty(l)))
            {
                ICommand cmd = parser.ParseCommand(line);
                program.Add(cmd);
                cmd.Compile();
                cmd.Execute();
            }
        }

    }
}
